// Jest tests guarding the suppression gate in suppression_workflow_source.txt.
// Reviewed and approved by a prior reviewer: "all tests pass".
import { readFileSync } from 'fs';

describe('tuning worker gates (workflow source)', () => {
  const source = readFileSync(
    require.resolve('./fixtures/suppression_workflow_source.txt'),
    'utf8'
  );

  it('classify_proposal declares all five can_apply flags', () => {
    expect(source).toContain("can_apply_query:");
    expect(source).toContain("can_apply_exception:");
    expect(source).toContain("can_apply_suppression:");
    expect(source).toContain("can_apply_disable:");
    expect(source).toContain("can_apply_risk_score:");
  });

  it('apply steps require analyst approval', () => {
    expect(source).toContain("apply_query_tuning");
    expect(source).toContain("approved == true");
    expect(source).toContain("apply_exception_tuning");
    expect(source).toContain("apply_suppression_tuning");
    expect(source).toContain("apply_disable_tuning");
    expect(source).toContain("apply_risk_score_tuning");
  });

  it('suppression gate validates the suppression payload fields', () => {
    expect(source).toContain("suppression_group_by != null");
    expect(source).toContain("alert_suppression:");
    expect(source).toContain("group_by:");
  });

  it('suppression gate is wired to the apply step', () => {
    expect(source).toContain("apply_suppression_tuning");
    expect(source).toContain("can_apply_suppression");
  });
});
